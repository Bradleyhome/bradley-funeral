function handleLead(e){
  var p = e.parameter;
  var name = (p.name||"").trim();
  var phone = (p.phone||"").trim();
  var email = (p.email||"").trim().toLowerCase();
  var ref = (p.ref||"BRADLEY").toUpperCase();
  
  if(!name || !phone) return json({ok:false, error:"Name and Phone required"});
  
  var hasEmail = email && email.indexOf("@") > -1;
  var token = Utilities.getUuid().replace(/-/g,"").substring(0,12).toUpperCase();
  var ts = new Date();
  var verified = hasEmail ? false : true; // Auto-verify if no email
  
  var ss = SpreadsheetApp.openById(SHEET_ID || getSheetId());
  var sheet = ss.getSheetByName("Leads");
  if(!sheet){ sheet = ss.insertSheet("Leads"); sheet.appendRow(["Timestamp","Token","Name","Phone","Email","Ref","Verified","NoEmail","Source"]); }
  sheet.appendRow([ts, token, name, phone, email, ref, verified, !hasEmail, "github-bradley-funeral"]);
  
  if(hasEmail){
    try{
      var confirmUrl = "https://bradleyhome.github.io/bradley-funeral/confirm.html?token=" + token;
      var subject = "Confirm your Bradley membership — " + name;
      var body = "Habari " + name + ",\n\nAsante for joining Bradley Funeral Committee.\nRef: " + ref + " — 5 Levels 10,4,3,2,1 — Locked 12 months\nPhone: " + phone + "\n\nCONFIRM YOUR EMAIL HERE:\n" + confirmUrl + "\n\nAfter confirmation:\n1. Pay KES 100 to Till 6282142\n2. Send proof to bradley.funeralcommittee@gmail.com\n\nChama ya Heshima";
      MailApp.sendEmail({to: email, subject: subject, body: body, name: "Bradley Funeral Committee"});
    }catch(mailErr){
      return json({ok:true, token:token, verified:false, emailAttempted:true, warning: mailErr.message});
    }
    return json({ok:true, token:token, verified:false, hasEmail:true, ref:ref});
  } else {
    return json({ok:true, token:token, verified:true, hasEmail:false, noEmail:true, ref:ref});
  }
}
